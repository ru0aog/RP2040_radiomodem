# 1. Полностью очищаем внутренний кэш отслеживания Git
git rm -r --cached .

# 2. Начисто сканируем всю папку на диске и добавляем ВСЕ файлы заново
git add --all

# 3. Создаем коммит тотальной локальной версии
git commit -m "Total enforce: overwrite GitHub with 100% local state"

# 4. Принудительно выталкиваем всё на GitHub, стирая старую историю на сервере
git push origin main --force
